123void gui::Button::update(const sf::Vector2i& mousePosWindow)
124{
125 // std::cout << "Position de la souris " << mousePosWindow.x << " " << mousePosWindow.y << std::endl;
126 // Idle
127 this->buttonState = BTN_IDLE;
128
129 // Hover
130 if (this->shape.getGlobalBounds().contains(static_cast<sf::Vector2f>(mousePosWindow)))
131 {
132 this->buttonState = BTN_HOVER;
133
134 // Pressed
135 if (sf::Mouse::isButtonPressed(sf::Mouse::Left))
136 {
137 this->buttonState = BTN_ACTIVE;
138 }
139 }
140
141 switch (this->buttonState)
142 {
143 case BTN_IDLE:
144 this->shape.setFillColor(this->idleColor);
145 this->text.setFillColor(this->textIdleColor);
146 this->shape.setOutlineColor(this->outlineIdleColor);
147 break;
148
149 case BTN_HOVER:
150 this->shape.setFillColor(this->hoverColor);
151 this->text.setFillColor(this->textHoverColor);
152 this->shape.setOutlineColor(this->outlineHoverColor);
153 break;
154
155 case BTN_ACTIVE:
156 this->shape.setFillColor(this->activeColor);
157 this->text.setFillColor(this->textActiveColor);
158 this->shape.setOutlineColor(this->outlineActiveColor);
159 break;
160
161 default:
162 this->shape.setFillColor(sf::Color::Red);
163 this->text.setFillColor(sf::Color::Blue);
164 this->shape.setOutlineColor(sf::Color::Green);
165 break;
166 }
167}
